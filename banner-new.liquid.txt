It sounds like you want to ensure the slideshow is fully swipeable on both **mobile touchscreens** and **desktop (using mouse drag)**, and that swiping left/right correctly navigates to the next or previous images. 

The previous code had basic touch support, but desktop mouse-dragging wasn't enabled, and sometimes the browser's default image-dragging behavior can block the swipe.

I have updated the JavaScript section below to include **Advanced Swipe Handling**. Now:
1. **Desktop users** can click and drag the banner left or right with their mouse to change slides.
2. **Mobile users** get a much more reliable touchscreen swipe experience.
3. If you are swiping, it will cancel the "click" so you don't accidentally open the banner link.

<script>
(function() {
  'use strict';
  
  var HeroSlideshow = {
    currentIndex: 0,
    totalSlides: 0,
    autoplayInterval: null,
    transitionSpeed: {{ section.settings.transition_speed | times: 1000 }},
    autoplaySpeed: {{ section.settings.autoplay_speed | times: 1000 }},
    isAutoplay: {{ section.settings.enable_autoplay }},
    isTransitioning: false,
    isPaused: false,
    
    init: function() {
      this.slides = document.querySelectorAll('.hero-slide');
      this.progressItems = document.querySelectorAll('.hero-progress-item');
      this.prevBtn = document.querySelector('.hero-nav-prev');
      this.nextBtn = document.querySelector('.hero-nav-next');
      
      if (!this.slides.length) return;
      
      this.totalSlides = this.slides.length;
      
      // Show first slide immediately
      var firstSlide = this.slides[0];
      firstSlide.classList.add('active');
      
      // Initialize progress bars
      this.resetAllProgressBars();
      if (this.progressItems[0]) {
        this.progressItems[0].classList.add('active');
        var firstBar = this.progressItems[0].querySelector('.hero-progress-bar');
        if (firstBar) {
          var self = this;
          setTimeout(function() {
            firstBar.style.animation = 'fillProgress ' + (self.autoplaySpeed / 1000) + 's linear forwards';
          }, 100);
        }
      }
      
      // Setup navigation
      if (this.prevBtn) {
        this.prevBtn.addEventListener('click', this.handlePrev.bind(this));
      }
      if (this.nextBtn) {
        this.nextBtn.addEventListener('click', this.handleNext.bind(this));
      }
      
      // Setup progress bar clicks
      for (var i = 0; i < this.progressItems.length; i++) {
        this.progressItems[i].addEventListener('click', this.goToSlide.bind(this, i));
      }
      
      // Keyboard navigation
      document.addEventListener('keydown', this.handleKeyboard.bind(this));
      
      // Touch & Mouse Swipe setup
      this.setupAdvancedSwipe();
      
      // Pause on hover
      var slideshow = document.querySelector('.hero-slideshow');
      if (slideshow) {
        slideshow.addEventListener('mouseenter', this.pause.bind(this));
        slideshow.addEventListener('mouseleave', this.resume.bind(this));
      }
      
      // Start autoplay
      if (this.isAutoplay && this.totalSlides > 1) {
        this.startAutoplay();
      }
    },
    
    resetAllProgressBars: function() {
      for (var i = 0; i < this.progressItems.length; i++) {
        this.progressItems[i].classList.remove('active', 'completed');
        var bar = this.progressItems[i].querySelector('.hero-progress-bar');
        if (bar) {
          bar.style.animation = 'none';
          bar.style.width = '0%';
          void bar.offsetWidth;
        }
      }
    },
    
    updateProgressBars: function(newIndex) {
      this.resetAllProgressBars();
      
      for (var i = 0; i < newIndex; i++) {
        if (this.progressItems[i]) {
          this.progressItems[i].classList.add('completed');
        }
      }
      
      if (this.progressItems[newIndex]) {
        var currentProgress = this.progressItems[newIndex];
        currentProgress.classList.add('active');
        
        var bar = currentProgress.querySelector('.hero-progress-bar');
        if (bar) {
          void bar.offsetWidth;
          bar.style.animation = 'fillProgress ' + (this.autoplaySpeed / 1000) + 's linear forwards';
        }
      }
    },
    
    showSlide: function(newIndex) {
      if (newIndex === this.currentIndex || this.isTransitioning) return;
      
      this.isTransitioning = true;
      
      var currentSlide = this.slides[this.currentIndex];
      var nextSlide = this.slides[newIndex];
      
      this.updateProgressBars(newIndex);
      
      currentSlide.classList.remove('active');
      currentSlide.classList.add('previous');
      
      nextSlide.classList.add('active');
      
      var self = this;
      setTimeout(function() {
        currentSlide.classList.remove('previous');
        
        self.currentIndex = newIndex;
        self.isTransitioning = false;
      }, this.transitionSpeed);
    },
    
    nextSlide: function() {
      var nextIndex = (this.currentIndex + 1) % this.totalSlides;
      this.showSlide(nextIndex);
    },
    
    handleNext: function() {
      this.stopAutoplay();
      this.nextSlide();
      if (this.isAutoplay && this.totalSlides > 1) {
        this.startAutoplay();
      }
    },
    
    handlePrev: function() {
      this.stopAutoplay();
      var prevIndex = (this.currentIndex - 1 + this.totalSlides) % this.totalSlides;
      this.showSlide(prevIndex);
      if (this.isAutoplay && this.totalSlides > 1) {
        this.startAutoplay();
      }
    },
    
    goToSlide: function(index) {
      if (index !== this.currentIndex && !this.isTransitioning) {
        this.stopAutoplay();
        this.showSlide(index);
        if (this.isAutoplay && this.totalSlides > 1) {
          this.startAutoplay();
        }
      }
    },
    
    handleKeyboard: function(e) {
      if (e.key === 'ArrowLeft') {
        this.handlePrev();
      } else if (e.key === 'ArrowRight') {
        this.handleNext();
      }
    },
    
    setupAdvancedSwipe: function() {
      var self = this;
      var slideshow = document.querySelector('.hero-slideshow');
      if (!slideshow) return;
      
      var startX = 0;
      var startY = 0;
      var isDragging = false;
      var didSwipe = false;
      
      // --- Touch Events (Mobile) ---
      slideshow.addEventListener('touchstart', function(e) {
        startX = e.touches[0].clientX;
        startY = e.touches[0].clientY;
        didSwipe = false;
      }, { passive: true });
      
      slideshow.addEventListener('touchend', function(e) {
        var endX = e.changedTouches[0].clientX;
        var endY = e.changedTouches[0].clientY;
        self.handleSwipeLogic(startX, startY, endX, endY);
      }, { passive: true });

      // --- Mouse Events (Desktop) ---
      slideshow.addEventListener('mousedown', function(e) {
        // Only allow left-click drags
        if (e.button !== 0) return; 
        isDragging = true;
        didSwipe = false;
        startX = e.clientX;
        startY = e.clientY;
      });

      slideshow.addEventListener('mousemove', function(e) {
        if (!isDragging) return;
        var diffX = Math.abs(startX - e.clientX);
        var diffY = Math.abs(startY - e.clientY);
        
        // If they drag more than 10 pixels, treat it as a swipe attempt (not a click)
        if (diffX > 10 || diffY > 10) {
          didSwipe = true;
        }
      });

      slideshow.addEventListener('mouseup', function(e) {
        if (!isDragging) return;
        isDragging = false;
        var endX = e.clientX;
        var endY = e.clientY;
        if (didSwipe) {
          self.handleSwipeLogic(startX, startY, endX, endY);
        }
      });

      slideshow.addEventListener('mouseleave', function(e) {
        if (isDragging) {
          isDragging = false;
          var endX = e.clientX;
          var endY = e.clientY;
          if (didSwipe) {
             self.handleSwipeLogic(startX, startY, endX, endY);
          }
        }
      });

      // Prevent the banner link from opening if the user was just dragging to swipe
      slideshow.addEventListener('click', function(e) {
        if (didSwipe) {
          e.preventDefault();
          e.stopPropagation();
        }
      }, true);
      
      // Prevent browser from default "ghost dragging" the banner images
      slideshow.addEventListener('dragstart', function(e) {
        e.preventDefault();
      });
    },

    handleSwipeLogic: function(startX, startY, endX, endY) {
      var diffX = startX - endX;
      var diffY = startY - endY;
      
      // Require a minimum swipe distance of 40px and ensure it's mostly horizontal
      if (Math.abs(diffX) > Math.abs(diffY) && Math.abs(diffX) > 40) {
        if (diffX > 0) {
          this.handleNext(); // User swiped left <- (shows the next image)
        } else {
          this.handlePrev(); // User swiped right -> (shows the previous image)
        }
      }
    },
    
    startAutoplay: function() {
      if (!this.isAutoplay || this.totalSlides <= 1 || this.isPaused) return;
      
      this.stopAutoplay();
      
      var self = this;
      this.autoplayInterval = setInterval(function() {
        if (!self.isPaused && !self.isTransitioning) {
          self.nextSlide();
        }
      }, this.autoplaySpeed);
    },
    
    stopAutoplay: function() {
      if (this.autoplayInterval) {
        clearInterval(this.autoplayInterval);
        this.autoplayInterval = null;
      }
    },
    
    pause: function() {
      this.isPaused = true;
      this.stopAutoplay();
    },
    
    resume: function() {
      this.isPaused = false;
      if (this.isAutoplay && this.totalSlides > 1) {
        this.startAutoplay();
      }
    }
  };
  
  if (document.readyState === 'loading') {
    document.addEventListener('DOMContentLoaded', function() {
      HeroSlideshow.init();
    });
  } else {
    HeroSlideshow.init();
  }
  
  if (window.Shopify && window.Shopify.designMode) {
    document.addEventListener('shopify:section:load', function() {
      setTimeout(function() {
        HeroSlideshow.init();
      }, 100);
    });
    
    document.addEventListener('shopify:section:unload', function() {
      HeroSlideshow.stopAutoplay();
    });
    
    document.addEventListener('shopify:block:select', function() {
      HeroSlideshow.stopAutoplay();
    });
  }
})();
